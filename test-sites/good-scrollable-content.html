function calculateComplexAlgorithm(data) {
    const result = [];
    
    for (let i = 0; i < data.length; i++) {
        const item = data[i];
        
        if (item.type === 'special') {
            result.push({
                id: item.id,
                value: item.value * 2,
                timestamp: new Date().toISOString(),
                metadata: {
                    processed: true,
                    processor: 'algorithm-v2',
                    confidence: 0.95
                }
            });
        } else {
            result.push({
                id: item.id,
                value: item.value,
                timestamp: new Date().toISOString(),
                metadata: {
                    processed: false,
                    processor: 'none',
                    confidence: 1.0
                }
            });
        }
    }
    
    return result.sort((a, b) => b.value - a.value);
}
